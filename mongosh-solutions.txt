    {
        $lookup: {
            from: "books",
            localField: "book_id",
            foreignField: "_id",
            as: "book_details"
        }
    },
    {
        $project: {
            _id: 0,
            action: 1,
            "book_details.title": 1,
            "book_details.author": 1,
            "book_details.year": 1
        }
    }
]).toArray();
